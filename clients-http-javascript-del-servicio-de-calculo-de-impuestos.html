    14.2.1. La capa [dao]

    La capa [dao] se implementará mediante la siguiente clase [Dao1.js]:

    
'use strict';
 
// importaciones
import qs from 'qs'
 
class Dao1 {
 
  // fabricante
  constructor(axios) {
    // biblioteca axios para realizar peticiones HTTP
    this.axios = axios;
    // cookie de sesión
    this.sessionCookieName = "PHPSESSID";
    this.sessionCookie = '';
  }
 
  // iniciar sesión
  async  initSession() {
    // opciones de consulta HHTP [get /main.php?action=init-session&type=json]
    const options = {
      method: "GET",
      // parámetros URL
      params: {
        action: 'init-session',
        type: 'json'
      }
    };
    // ejecución de la solicitud HTTP
    return await this.getRemoteData(options);
  }

  async  authentifierUtilisateur(user, password) {
    // opciones de consulta HHTP [post /main.php?action=authentifier-utilisateur]
    const options = {
      method: "POST",
      headers: {
        'Content-type": "application/x-www-form-urlencoded",
      },
      // cuerpo del POST
      data: qs.stringify({
        user: user,
        password: password
      }),
      // parámetros URL
      params: {
        action: 'authentifier-utilisateur'
      }
    };
    // ejecución de la solicitud HTTP
    return await this.getRemoteData(options);
  }
 
  // cálculo de impuestos
  async  calculerImpot(marié, enfants, salaire) {
    // opciones de consulta HHTP [post /main.php?action=calculer-impot]
    const options = {
      method: "POST",
      headers: {
        'Content-type": "application/x-www-form-urlencoded",
      },
      // cuerpo de POST [marié, enfants, salaire]
      data: qs.stringify({
        marié: marié,
        enfants: enfants,
        salaire: salaire
      }),
      // parámetros URL
      params: {
        action: 'calculer-impot'
      }
    };
    // ejecución de la solicitud HTTP
    const data = await this.getRemoteData(options);
    // resultado
    return data;
  }
 
  // lista de simulaciones
  async  listeSimulations() {
    // opciones de consulta HHTP [get /main.php?action=lister-simulations]
    const options = {
      method: "GET",
      // parámetros URL
      params: {
        action: 'lister-simulations'
      },
    };
    // ejecución de la solicitud HTTP
    const data = await this.getRemoteData(options);
    // resultado
    return data;
  }
 
  // lista de simulaciones
  async  supprimerSimulation(index) {
    // opciones de consulta HHTP [get /main.php?action=supprimer-simulation&numéro=index]
    const options = {
      method: "GET",
      // parámetros URL
      params: {
        action: 'supprimer-simulation',
        numéro: index
      },
    };
    // ejecución de la solicitud HTTP
    const data = await this.getRemoteData(options);
    // resultado
    return data;
  }
 
  async  getRemoteData(options) {
    // para la cookie de sesión
    if (!options.headers) {
      options.headers = {};
    }
    options.headers.Cookie = this.sessionCookie;
    // ejecución de la solicitud HTTP
    let response;
    try {
      // solicitud asíncrona
      response = await this.axios.request('main.php', options);
    } catch (error) {
      // el parámetro [error] es una instancia de excepción - puede tomar varias formas
      if (error.response) {
        // la respuesta del servidor está en [error.response]
        response = error.response;
      } else {
        // reiniciar el error
        throw error;
      }
    }
    // response es la respuesta HTTP completa del servidor (cabeceras HTTP + respuesta propiamente dicha)
    // recuperar la cookie de sesión si existe
    const setCookie = response.headers['set-cookie'];
    if (setCookie) {
      // setCookie es una matriz
      // buscar la cookie de sesión en esta tabla
      let trouvé = false;
      let i = 0;
      while (!trouvé && i < setCookie.length) {
        // buscar la cookie de sesión
        const results = RegExp('^(' + this.sessionCookieName + '.+?);').exec(setCookie[i]);
        if (results) {
          // guardamos la cookie de sesión
          // eslint-disable-next-line require-atomic-updates
          this.sessionCookie = results[1];
          // encontramos
          trouvé = true;
        } else {
          // punto siguiente
          i++;
        }
      }
    }
    // la respuesta del servidor está en [response.data]
    return response.data;
  }
}
 
// exportación de clases
export default Dao1;

    • líneas 9-15: el constructor de la clase. Esta tendrá tres propiedades:
      • [axios]: el objeto [axios] que permite realizar las consultas HTTP. Este es transmitido por el código que realiza la llamada;
      • [sessionCookieName]: dependiendo de los servidores, la cookie de sesión tiene diferentes nombres. En este caso, es [PHPSESSID];
      • [sessionCookie]: la cookie de sesión enviada por el servidor y almacenada por el cliente;
    • líneas 53-76: la función asíncrona [calculerImpot] realiza la solicitud [post /main.php?action=calculer-impot] enviando los parámetros [marié, enfants, salaire]. Devuelve la cadena jSON transmitida por el servidor en forma de un objeto Javascript;
    • líneas 79-92: la función asíncrona [listeSimulations] realiza la solicitud [get /main.php?action=lister-simulations. Devuelve la cadena jSON transmitida por el servidor en forma de un objeto Javascript;
    • líneas 95-109: la función asíncrona [supprimerSimulation] realiza la consulta [get /main.php?action=supprimer-simulation&numéro=index]. Devuelve la cadena jSON transmitida por el servidor en forma de un objeto Javascript;
    • línea 121: se utiliza la notación [this.axios] porque, en este caso, el objeto [axios] transmitido al constructor se ha almacenado en la propiedad [this.axios];
    • línea 161: la clase [Dao1] se exporta para poder ser utilizada;

    14.2.2. El script [main1.js]

    El script [main1.js] realiza una serie de llamadas al servidor utilizando la clase [Dao1]:

    • inicialización de una sesión jSON;
    • autenticación con [admin, admin];
    • solicita tres cálculos de impuestos;
    • solicita la lista de simulaciones;
    • elimina una de ellas;

    El código es el siguiente:

    
// importar axios
import axios from 'axios';
// importar clase Dao1
import Dao from './Dao1';
 
// función asíncrona [main]
async function main() {
  // configuración de axios
  axios.defaults.timeout = 2000;
  axios.defaults.baseURL = 'http://localhost/php7/scripts-web/impots/version-14';
  // capa de instanciación [dao]
  const dao = new Dao(axios);
  // utilización de la capa [dao]
  try {
    // iniciar sesión
    log("-----------init-session");
    let response = await dao.initSession();
    log(response);
    // autenticación
    log("-----------authentifier-utilisateur");
    response = await dao.authentifierUtilisateur("admin", "admin");
    log(response);
    // cálculo de impuestos
    log("-----------calculer-impot x 3");
    response = await Promise.all([
      dao.calculerImpot("oui", 2, 45000),
      dao.calculerImpot("non", 2, 45000),
      dao.calculerImpot("non", 1, 30000)
    ]);
    log(response);
    // lista de simulaciones
    log("-----------liste-des-simulations");
    response = await dao.listeSimulations();
    log(response);
    // borrar una simulación
    log("-----------suppression simulation n° 1");
    response = await dao.supprimerSimulation(1);
    log(response);
  } catch (error) {
    // registramos el error
    console.log("erreur=", error.message);
  }
}
 
// log jSON
function log(object) {
  console.log(JSON.stringify(object, null, 2));
}
 
// ejecución
main();

    14.3. Cliente HTTP 2

    La arquitectura del cliente HTTP2 es la siguiente:

    Se ha trasladado la capa [métier] del servidor al cliente Javascript. A diferencia de lo que hicimos en el curso PHP7, la capa [main] no tendrá que pasar aquí por la capa [métier] para llegar a la capa [dao]. Utilizaremos estas dos capas como centros de competencias:

    • la capa [main] pasa por la capa [dao] en cuanto necesita datos que se encuentran en el servidor;
    • la capa [main] solicita a la capa [métier] que realice los cálculos del impuesto;
    • la capa [métier] es independiente de la capa [dao] y nunca recurre a ella;

    14.3.1. La clase Javascript [Métier]

    La esencia de la clase [Métier] en PHP se ha descrito en el artículo enlace. Se trata de un código bastante complejo que recordamos aquí, no para explicarlo, sino para poder traducirlo a Javascript:

    
<?php
 
// espacio de nombres
namespace Application;
 
class Metier implements InterfaceMetier {
  // capa Dao
  private $dao;
  // datos de la administración fiscal
  private $taxAdminData;
 
  //---------------------------------------------
  // setter capa [dao]
  public function setDao(InterfaceDao $dao) {
    $this->dao = $dao;
    return $this;
  }
 
  public function __construct(InterfaceDao $dao) {
    // almacenamos una referencia en la capa [dao]
    $this->dao = $dao;
    // recuperamos los datos necesarios para calcular los impuestos
    // el método [getTaxAdminData] puede lanzar una excepción ExceptionImpots
    // luego dejamos que vuelva al código de llamada
    $this->taxAdminData = $this->dao->getTaxAdminData();
  }
 
// cálculo de impuestos
// --------------------------------------------------------------------------
  public function calculerImpot(string $marié, int $enfants, int $salaire): array {
    // $marié : sí, no
    // $enfants : número de hijos
    // $salaire: salario anual
    // $this->taxAdminData: datos de la administración fiscal
    //
    // comprobamos que tenemos los datos correctos de las autoridades fiscales
    if ($this->taxAdminData === NULL) {
      $this->taxAdminData = $this->getTaxAdminData();
    }
    // cálculo de impuestos con hijos
    $result1 = $this->calculerImpot2($marié, $enfants, $salaire);
    $impot1 = $result1["impôt"];
    // cálculo de impuestos sin hijos
    if ($enfants != 0) {
      $result2 = $this->calculerImpot2($marié, 0, $salaire);
      $impot2 = $result2["impôt"];
      // aplicación del límite máximo de las prestaciones familiares
      $plafonDemiPart = $this->taxAdminData->getPlafondQfDemiPart();
      if ($enfants < 3) {
        // $PLAFOND_QF_DEMI_PART euros para los 2 primeros niños
        $impot2 = $impot2 - $enfants * $plafonDemiPart;
      } else {
        // $PLAFOND_QF_DEMI_PART euros para los 2 primeros hijos, el doble para los hijos siguientes
        $impot2 = $impot2 - 2 * $plafonDemiPart - ($enfants - 2) * 2 * $plafonDemiPart;
      }
    } else {
      $impot2 = $impot1;
      $result2 = $result1;
    }
    // cobramos el impuesto más alto
    if ($impot1 > $impot2) {
      $impot = $impot1;
      $taux = $result1["taux"];
      $surcôte = $result1["surcôte"];
    } else {
      $surcôte = $impot2 - $impot1 + $result2["surcôte"];
      $impot = $impot2;
      $taux = $result2["taux"];
    }
    // cálculo de cualquier descuento
    $décôte = $this->getDecôte($marié, $salaire, $impot);
    $impot -= $décôte;
    // cálculo de cualquier reducción fiscal
    $réduction = $this->getRéduction($marié, $salaire, $enfants, $impot);
    $impot -= $réduction;
    // resultado
    return ["impôt" => floor($impot), "surcôte" => $surcôte, "décôte" => $décôte, "réduction" => $réduction, "taux" => $taux];
  }
 
// --------------------------------------------------------------------------
  private function calculerImpot2(string $marié, int $enfants, float $salaire): array {
    // $marié : sí, no
    // $enfants : número de hijos
    // $salaire: salario anual
    // $this->taxAdminData: datos de la administración fiscal
    //
    // número de acciones
    $marié = strtolower($marié);
    if ($marié === "oui") {
      $nbParts = $enfants / 2 + 2;
    } else {
      $nbParts = $enfants / 2 + 1;
    }
    // 1 parte por niño a partir del 3er
    if ($enfants >= 3) {
      // medio punto porcentual más por cada niño a partir del 3º
      $nbParts += 0.5 * ($enfants - 2);
    }
    // ingresos imponibles
    $revenuImposable = $this->getRevenuImposable($salaire);
    // surcôte
    $surcôte = floor($revenuImposable - 0.9 * $salaire);
    // para problemas de redondeo
    if ($surcôte < 0) {
      $surcôte = 0;
    }
    // cociente familiar
    $quotient = $revenuImposable / $nbParts;
    // cálculo de impuestos
    $limites = $this->taxAdminData->getLimites();
    $coeffR = $this->taxAdminData->getCoeffR();
    $coeffN = $this->taxAdminData->getCoeffN();
    // se establece al final de la tabla de límites para detener el bucle que sigue a
    $limites[count($limites) - 1] = $quotient;
    // búsqueda de tipos impositivos
    $i = 0;
    while ($quotient > $limites[$i]) {
      $i++;
    }
    // porque $quotient se ha colocado al final de la matriz $limites, el bucle anterior
    // no puede superar la tabla $limites
    // ahora podemos calcular el impuesto
    $impôt = floor($revenuImposable * $coeffR[$i] - $nbParts * $coeffN[$i]);
    // resultado
    return ["impôt" => $impôt, "surcôte" => $surcôte, "taux" => $coeffR[$i]];
  }
 
  // revenuImposable=AbonaciónSalarioAnual
  // la indemnización tiene un mínimo y un máximo
  private function getRevenuImposable(float $salaire): float {
    // deducción del 10% del salario
    $abattement = 0.1 * $salaire;
    // esta asignación no puede superar $this->taxAdminData->getAbattementDixPourCentMax()
    if ($abattement > $this->taxAdminData->getAbattementDixPourCentMax()) {
      $abattement = $this->taxAdminData->getAbattementDixPourcentMax();
    }
    // la rebaja no puede ser inferior a $this->taxAdminData->getAbattementDixPourcentMin()
    if ($abattement < $this->taxAdminData->getAbattementDixPourcentMin()) {
      $abattement = $this->taxAdminData->getAbattementDixPourcentMin();
    }
    // ingresos imponibles
    $revenuImposable = $salaire - $abattement;
    // resultado
    return floor($revenuImposable);
  }
 
// calcula cualquier descuento
  private function getDecôte(string $marié, float $salaire, float $impots): float {
    // al principio, un descuento cero
    $décôte = 0;
    // importe máximo del impuesto para beneficiarse del descuento
    $plafondImpôtPourDécôte = $marié === "oui" ?
      $this->taxAdminData->getPlafondImpotCouplePourDecote() :
      $this->taxAdminData->getPlafondImpotCelibatairePourDecote();
    if ($impots < $plafondImpôtPourDécôte) {
      // descuento máximo
      $plafondDécôte = $marié === "oui" ?
        $this->taxAdminData->getPlafondDecoteCouple() :
        $this->taxAdminData->getPlafondDecoteCelibataire();
      // descuento teórico
      $décôte = $plafondDécôte - 0.75 * $impots;

    El problema CORS proviene de la relación cliente/servidor:

    • nuestro cliente Javascript se ha descargado en la máquina [http://localhost:8080];
    • el servidor de cálculo de impuestos se ejecuta en la máquina [http://localhost:80];
    • el cliente y el servidor no se encuentran, por tanto, en los mismos dominios (mismo equipo, pero no el mismo puerto);
    • El navegador que ejecuta el cliente Javascript, cargado desde la máquina [http://localhost:8080], bloquea cualquier solicitud que no tenga como destino [http://localhost:80]. Se trata de una medida de seguridad. Por lo tanto, bloquea la solicitud del cliente al servidor que opera en la máquina [http://localhost:80];

    De hecho, el navegador no bloquea totalmente la solicitud. En realidad, espera a que el servidor le «diga» que acepta las solicitudes entre dominios. Si obtiene esta autorización, el navegador transmitirá entonces la solicitud entre dominios.

    El servidor da su autorización enviando encabezados HTTP específicos:

    • [3]: en las clases [HtmlResponse, JsonResponse, XmlResponse] que generan las respuestas para las sesiones [html, json, xml], respectivamente.     La clase [ParentResponse] [2] evoluciona de la siguiente manera:

    
<?php
 
namespace Application;
 
// dependencias de Symfony
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\HttpFoundation\Request;
 
class ParentResponse {
 
  // int $statusCode: el HTTP código de estado de la respuesta
  // string $content : el cuerpo de la respuesta a enviar
  // según el caso, se trata de una cadena JSON, XML, HTML
  // array $headers: las cabeceras HTTP que se añadirán a la respuesta
 
  public function sendResponse(
    Request $request,
    int $statusCode,
    string $content,
    array $headers,
    array $config): void {
 
    // preparar el texto de respuesta del servidor
    $response = new Response();
    $response->setCharset("utf-8");
    // código de situación
    $response->setStatusCode($statusCode);
    // cabeceras para solicitudes entre dominios
    if ($config['corsAllowed']) {
      $origin = $request->headers->get("origin");
      if (strpos($origin, "http://localhost") === 0) {
        $headers = array_merge($headers,
          ["Access-Control-Allow-Origin" => $origin,
            "Access-Control-Allow-Headers" => "Accept, Content-Type",
            "Access-Control-Allow-Methods" => "GET, POST",
            "Access-Control-Allow-Credentials" => "true"
        ]);
      }
    }
    foreach ($headers as $text => $value) {
      $response->headers->set($text, $value);
    }
    // caso especial del método [OPTIONS]
    // en este caso sólo son importantes las cabeceras
    $method = strtolower($request->getMethod());
    if ($method === "options") {
      $content = "";
      $response->setStatusCode(Response::HTTP_OK);
    }
    // enviamos la respuesta
    $response->setContent($content);
    $response->send();
  }
 
}

    14.5. Mejora del cliente HTTP 3

    Cuando la clase [Dao2] anterior se ejecuta en un navegador, la gestión de la cookie de sesión es innecesaria. De hecho, es el navegador que aloja la capa [dao] el que gestiona la cookie de sesión: devuelve automáticamente cualquier cookie que le envíe el servidor. "use strict";
 
// importaciones
import qs from "qs";
 
class Dao3 {
  // fabricante
  constructor(axios) {
    this.axios = axios;
  }
 
  // iniciar sesión
  async initSession() {
    // opciones de consulta HHTP [get /main.php?action=init-session&type=json]
    const options = {
      method: "GET",
      // parámetros URL
      params: {
        action: "init-session",
        type: "json"
      }
    };
    // ejecución de la solicitud HTTP
    return await this.getRemoteData(options);
  }
 
  async authentifierUtilisateur(user, password) {
    // opciones de consulta HHTP [post /main.php?action=authentifier-utilisateur]
    const options = {
      method: "POST",
      headers: {
        "Content-type": "application/x-www-form-urlencoded"
      },
      // cuerpo del POST
      data: qs.stringify({
        user: user,
        password: password
      }),
      // parámetros URL
      params: {
        action: "authentifier-utilisateur"
      }
    };
    // ejecución de la solicitud HTTP
    return await this.getRemoteData(options);
  }
 
  async getAdminData() {
    // opciones de consulta HHTP [get /main.php?action=get-admindata]
    const options = {
      method: "GET",
      // parámetros URL
      params: {
        action: "get-admindata"
      }
    };
    // ejecución de la solicitud HTTP
    const data = await this.getRemoteData(options);
    // resultado
    return data;
  }
 
  async getRemoteData(options) {
    // ejecución de la solicitud HTTP
    let response;
    try {
      // solicitud asíncrona
      response = await this.axios.request("main.php", options);
    } catch (error) {
      // el parámetro [error] es una instancia de excepción - puede tomar varias formas
      if (error.response) {
        // la respuesta del servidor está en [error.response]
        response = error.response;
      } else {
        // reiniciar el error
        throw error;
      }
    }
    // response es la respuesta HTTP completa del servidor (cabeceras HTTP + respuesta propiamente dicha)
    // la respuesta del servidor está en [response.data]
    return response.data;
  }
}
